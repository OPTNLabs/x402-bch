cmake_minimum_required(VERSION 3.16)
project(x402_bch LANGUAGES CXX)
include(GNUInstallDirs)

add_library(x402_bch STATIC src/server.cpp src/sdk.cpp src/sdk_wallet.cpp src/sdk_builders.cpp src/sdk_provider.cpp src/sdk_roles.cpp src/covenant.cpp src/sdk_mocknet.cpp src/diagnostics.cpp)
target_include_directories(x402_bch PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(x402_bch PUBLIC cxx_std_17)
find_library(GMPXX_LIBRARY NAMES gmpxx REQUIRED)
find_library(GMP_LIBRARY NAMES gmp REQUIRED)
find_library(SODIUM_LIBRARY NAMES sodium REQUIRED)
target_link_libraries(x402_bch PUBLIC ${GMPXX_LIBRARY} ${GMP_LIBRARY} ${SODIUM_LIBRARY})
target_compile_definitions(x402_bch PRIVATE X402_BCH_BIP39_WORDLIST_PATH="${CMAKE_INSTALL_FULL_DATADIR}/x402-bch/bip39-english.txt")

add_executable(x402_bch_conformance tests/conformance_test.cpp)
target_link_libraries(x402_bch_conformance PRIVATE x402_bch)
add_executable(x402_bch_sdk_test tests/sdk_test.cpp)
target_link_libraries(x402_bch_sdk_test PRIVATE x402_bch)
enable_testing()
add_test(NAME x402_bch_conformance COMMAND x402_bch_conformance WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME x402_bch_sdk_test COMMAND x402_bch_sdk_test WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})

install(TARGETS x402_bch ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES include/x402_bch.hpp include/x402_bch_sdk.hpp DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/x402-bch)
install(DIRECTORY include/x402_bch DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES spec/conformance/v1/bip39-english.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/x402-bch)
